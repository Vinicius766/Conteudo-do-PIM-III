Código banco de dados

/* Lógico3_PIM: */

CREATE TABLE CLIENTES (
    CPF CHAR(11) PRIMARY KEY,
    NOME VARCHAR(100),
    EMAIL VARCHAR(50),
    TELEFONE VARCHAR(15),
    DATA_NASCIMENTO DATE,
    OBJETIVO_NUTRICIONAL VARCHAR(255)
);

CREATE TABLE ENDERECO (
    ENDERECO_ID INT PRIMARY KEY,
    RUA VARCHAR(100),
    NUMERO VARCHAR(10),
    BAIRRO VARCHAR(50),
    CEP VARCHAR(9),
    COMPLEMENTO VARCHAR(50)
);

CREATE TABLE PEDIDOS (
    PEDIDO_ID INT PRIMARY KEY,
    DATA_HORA DATETIME,
    VALOR_TOTAL DECIMAL(10,2),
    STATUS_PEDIDO VARCHAR(30),
    FORMA_PAGAMENTO VARCHAR(30),
    FK_CLIENTE_CPF CHAR(11)
);

CREATE TABLE ITENS_PEDIDOS (
    ITEM_ID INT PRIMARY KEY,
    QUANTIDADE INT,
    PRECO_NO_MOMENTO DECIMAL(10,2),
    FK_PEDIDOS_ID INT,
    FK_PRODUTOS_INFO_NUTRICIONAL_PRODUTO_ID INT
);

CREATE TABLE PRODUTOS_INFO_NUTRICIONAL (
    PRODUTO_ID INT PRIMARY KEY,
    NOME_PRODUTO VARCHAR(100),
    DESCRICAO VARCHAR(255),
    PRECO DECIMAL(10,2),
    CATEGORIA VARCHAR(50),
    QTD_ESTOQUE INT,
    NUTRI_ID INT,
    CALORIAS DECIMAL(8,2),
    PROTEINAS DECIMAL(8,2),
    CARBOIDRATOS DECIMAL(8,2),
    GORDURAS_TOTAIS DECIMAL(8,2)
);

CREATE TABLE POSSUI (
    FK_CLIENTES_CPF CHAR(11),
    FK_ENDERECO_ID INT
);
 
ALTER TABLE PEDIDOS ADD CONSTRAINT FK_PEDIDOS_2
    FOREIGN KEY (FK_CLIENTE_CPF)
    REFERENCES CLIENTES (CPF);

 
ALTER TABLE ITENS_PEDIDOS ADD CONSTRAINT FK_ITENS_PEDIDOS_2
    FOREIGN KEY (FK_PEDIDOS_ID)
    REFERENCES PEDIDOS (PEDIDO_ID);

 
ALTER TABLE ITENS_PEDIDOS ADD CONSTRAINT FK_ITENS_PEDIDOS_3
    FOREIGN KEY (FK_PRODUTOS_INFO_NUTRICIONAL_PRODUTO_ID)
    REFERENCES PRODUTOS_INFO_NUTRICIONAL (PRODUTO_ID);
